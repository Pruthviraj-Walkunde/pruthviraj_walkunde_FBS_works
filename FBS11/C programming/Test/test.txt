int getcount(char* str){
	int i=0,count=1;
	while(str[i]!='\0'){
		if(str[i]==' '){
			count++;
		}
		i++;
	}
	return i;
}

#include<stdio.h>
struct song{
	int song_id;
	char song_title[30];
	char song_art[30];
	int song_len;
	struct song*next;
};
struct song *songs=NULL;
struct song*temp;
void add(){
	if(songs==NULL){
		songs=(struct song*)malloc(sizeof(struct song));
		songs->next=NULL;
		printf("Enter song ID:");
		scanf("%d",&songs->song_id);
		printf("Enter song TITLE:");
		scanf("%s",songs->song_title);
		printf("Enter song Artist:");
		scanf("%s",songs->song_art);
		printf("Enter song Length in (min,sec):");
		scanf("%d",&songs->song_len);
	
	}
	else{
		temp=(struct song*)malloc(sizeof(struct song));
		temp->next=NULL;
		next=temp;
		
		printf("Enter song ID:");
		scanf("%d",&next->song_id);
		printf("Enter song TITLE:");
		scanf("%s",next->song_title);
		printf("Enter song Artist:");
		scanf("%s",next->song_art);
		printf("Enter song Length in (min,sec):")
		scanf("%d",&next->song_len);
		
	}
	
	void display(){
		if(songs==NULL){
			printf("Not song found..");
		}
		else{
			printf("display");
		}
	}
